# Distill Electron Frontend Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the Distill Electron desktop app with electron-vite, React 19, TypeScript strict, TanStack Router, TanStack Query (useSuspenseQuery + Orval-generated functions), Zustand, GSAP/Anime.js animations, and a 3-panel Transcription view — integrated with the FastAPI backend at localhost:8000.

**Architecture:** Feature-based structure under `src/renderer/src/features/`, flat `ui/` design system, Electron main process auto-spawns uvicorn and polls `/health`, renderer talks directly to `http://localhost:8000` via Orval-generated Axios hooks (never via IPC). Zustand holds UI/process state; React Query holds all server state.

**Tech Stack:** Electron 33, electron-vite, React 19, TypeScript strict, Tailwind CSS 4, shadcn/ui, TanStack Router (file-based), TanStack Query (useSuspenseQuery), TanStack Virtual, Zustand, Axios, Orval, react-resizable-panels, GSAP, Anime.js, Vitest, Testing Library, MSW

```typescript
import { spawn, ChildProcess } from 'child_process'
import { EventEmitter } from 'events'
import { ipcMain } from 'electron'

export type BackendStatus = 'starting' | 'ready' | 'error' | 'fatal'

export class BackendManager extends EventEmitter {
  status: BackendStatus = 'starting'
  private process: ChildProcess | null = null
  private restartCount = 0
  private readonly maxRestarts = 3
  private healthTimer: NodeJS.Timeout | null = null

  setStatus(s: BackendStatus) {
    this.status = s
    this.emit('status', s)
  }

  start(webContents?: Electron.WebContents) {
    this.process = spawn('poetry', ['run', 'uvicorn', 'app.main:app', '--port', '8000'], {
      cwd: process.env.BACKEND_PATH ?? './backend',
      stdio: 'pipe',
    })

    this.process.on('exit', (code) => {
      if (code !== 0 && code !== null) {
        this.handleCrash(webContents)
      }
    })

    this.pollHealth(webContents)

    ipcMain.handle('backend:port', () => 8000)
  }

  private async pollHealth(webContents?: Electron.WebContents, elapsed = 0) {
    if (elapsed > 30000) {
      this.setStatus('error')
      webContents?.send('backend:status', 'error')
      return
    }

    try {
      const res = await fetch('http://localhost:8000/health')
      if (res.ok) {
        this.setStatus('ready')
        webContents?.send('backend:status', 'ready')
        return
      }
    } catch {
      // not ready yet
    }

    this.healthTimer = setTimeout(() => this.pollHealth(webContents, elapsed + 500), 500)
  }

  private handleCrash(webContents?: Electron.WebContents) {
    this.restartCount++
    if (this.restartCount > this.maxRestarts) {
      this.setStatus('fatal')
      webContents?.send('backend:fatal')
      return
    }
    this.setStatus('starting')
    webContents?.send('backend:status', 'starting')
    this.start(webContents)
  }

  stop() {
    if (this.healthTimer) clearTimeout(this.healthTimer)
    if (this.process) {
      this.process.kill('SIGTERM')
      setTimeout(() => this.process?.kill('SIGKILL'), 3000)
    }
  }
}

export const backendManager = new BackendManager()
```

```typescript
export type WsEvent = {
  type: string
  data: Record<string, unknown>
}

type WsCallback = (event: WsEvent) => void

class WsManager {
  private connections = new Map<string, WebSocket>()

  connect(transcriptionId: string, onEvent: WsCallback): WebSocket {
    if (this.connections.has(transcriptionId)) {
      return this.connections.get(transcriptionId)!
    }

    const ws = new WebSocket(`ws://localhost:8000/ws/${transcriptionId}`)

    ws.onmessage = (e) => {
      try {
        const event = JSON.parse(e.data) as WsEvent
        onEvent(event)
        if (event.type === 'pipeline_complete' || event.type === 'pipeline_error') {
          this.disconnect(transcriptionId)
        }
      } catch {
        // ignore malformed messages
      }
    }

    ws.onerror = () => {
      onEvent({ type: 'pipeline_error', data: { error: 'WebSocket error' } })
      this.disconnect(transcriptionId)
    }

    this.connections.set(transcriptionId, ws)
    return ws
  }

  disconnect(transcriptionId: string) {
    const ws = this.connections.get(transcriptionId)
    if (ws) {
      ws.close()
      this.connections.delete(transcriptionId)
    }
  }
}

export const wsManager = new WsManager()
```

    },

    }
  }, [isRecording, tickElapsed, setAudioLevel])

    elapsedSeconds,
    audioLevel,

export const Waveform = memo(function Waveform({ audioLevel, isRecording }: WaveformProps) {
  const barsRef = useRef<HTMLDivElement[]>([])
  const BAR_COUNT = 40

  useEffect(() => {
    if (!isRecording) {
      gsap.to(barsRef.current, { scaleY: 0.1, duration: 0.3 })
      return
    }

    barsRef.current.forEach((bar, i) => {
      const variance = Math.sin(Date.now() / 200 + i * 0.5) * 0.5 + 0.5

    })
  }, [audioLevel, isRecording])

    <div className="flex items-center gap-[3px] h-20">
      {Array.from({ length: BAR_COUNT }).map((_, i) => (
        <div
          key={i}
          ref={(el) => { if (el) barsRef.current[i] = el }}

import { wsManager } from '@renderer/lib/ws'

export function useImport() {
  const navigate = useNavigate()
  const queryClient = useQueryClient()
  const [url, setUrl] = useState('')
  const [urlError, setUrlError] = useState<string | null>(null)
  const { setCurrentId, setStatus, setProgress, reset } = useTranscriptionStore.getState()
  const progress = useTranscriptionStore((s) => s.progress)
  const status = useTranscriptionStore((s) => s.status)

  const isValidYouTubeUrl = (u: string) =>
    /^https?:\/\/(www\.)?(youtube\.com\/watch\?v=|youtu\.be\/)[\w-]+/.test(u)

  const importMutation = useMutation({
    mutationFn: () => postApiV1YoutubeProcess({ url }),
    onSuccess: (data) => {
      setCurrentId(data.transcription_id)
      setStatus('downloading')

      wsManager.connect(data.transcription_id, (event) => {
        if (event.type === 'youtube_download_start') setStatus('downloading')
        if (event.type === 'transcription_start') setStatus('transcribing')
        if (event.type === 'transcription_progress') setProgress((event.data.progress as number) ?? 0)
        if (event.type === 'pipeline_complete') {
          queryClient.invalidateQueries({ queryKey: ['transcriptions'] })
          navigate({ to: '/transcription/$id', params: { id: data.transcription_id } })
          reset()
        }
        if (event.type === 'pipeline_error') {
          setStatus('failed')
        }
      })
    },
  })

  const submit = () => {
    if (!isValidYouTubeUrl(url)) {
      setUrlError('Please enter a valid YouTube URL')
      return
    }
    setUrlError(null)
    importMutation.mutate()
  }

  return { url, setUrl, urlError, submit, progress, status, isPending: importMutation.isPending }
}

        llm_model: model,

    },
    onSuccess: (data) => {

    },
  })

  const send = () => {
    if (!input.trim() || sendMutation.isPending) return
    sendMutation.mutate(input.trim())
  }

  return { messages, input, setInput, send, isPending: sendMutation.isPending }

  const keyLabel: Record<string, { hasKey: boolean; placeholder: string }> = {
    openai: { hasKey: hasOpenAIKey, placeholder: 'sk-...' },
    gemini: { hasKey: hasGoogleKey, placeholder: 'AIza...' },
    anthropic: { hasKey: hasAnthropicKey, placeholder: 'sk-ant-...' },
  }

  const isCloud = provider !== 'ollama'

  return (
    <div className="space-y-5 max-w-md">
      <div className="space-y-2">
        <label className="text-sm text-white/50">Provider</label>
        <div className="flex gap-2">
          {PROVIDERS.map((p) => (
            <button
              key={p}
              onClick={() => setProvider(p)}
              className={`px-3 py-1.5 text-sm rounded-lg capitalize transition-colors ${
                provider === p ? 'bg-[var(--color-accent)] text-white' : 'bg-white/5 text-white/50 hover:bg-white/10'
              }`}
            >
              {p}
            </button>
          ))}
        </div>
      </div>

      {provider === 'ollama' ? (
        <div className="space-y-2">
          <label className="text-sm text-white/50">Model</label>
          <select
            value={model}
            onChange={(e) => setModel(e.target.value)}
            className="w-full px-3 py-2 bg-white/5 border border-white/10 rounded-lg text-sm outline-none"
          >
            {ollamaModels.map((m) => (
              <option key={m.name} value={m.name}>{m.name}</option>
            ))}
          </select>
        </div>
      ) : (
        <>
          <div className="space-y-2">
            <label className="text-sm text-white/50">Model</label>
            <input
              type="text"
              value={model}
              onChange={(e) => setModel(e.target.value)}
              placeholder={provider === 'openai' ? 'gpt-4o' : provider === 'gemini' ? 'gemini-2.0-flash' : 'claude-opus-4-7'}
              className="w-full px-3 py-2 bg-white/5 border border-white/10 rounded-lg text-sm outline-none"
            />
          </div>
          <div className="space-y-2">
            <label className="text-sm text-white/50">
              API Key {keyLabel[provider]?.hasKey && <span className="text-green-400 text-xs">✓ saved</span>}
            </label>
            <input
              type="password"
              value={apiKey}
              onChange={(e) => setApiKey(e.target.value)}
              placeholder={keyLabel[provider]?.placeholder ?? ''}
              className="w-full px-3 py-2 bg-white/5 border border-white/10 rounded-lg text-sm outline-none font-mono"
            />
          </div>
        </>
      )}

      <button
        onClick={() => onSave({ llm: { provider, model } })}
        disabled={isSaving}
        className="px-4 py-2 bg-[var(--color-accent)] text-white text-sm rounded-lg disabled:opacity-50 hover:opacity-90 transition-opacity"
      >
        {isSaving ? 'Saving...' : 'Save'}
      </button>
    </div>
  )
}

import { useSuspenseQuery, useMutation, useQueryClient } from '@tanstack/react-query'
import { getApiV1OllamaStatus, postApiV1OllamaStart, postApiV1OllamaStop } from '@renderer/lib/api/generated/ollama'
import { Circle } from 'lucide-react'

export function SystemStatus() {
  const queryClient = useQueryClient()

  const { data: ollamaStatus } = useSuspenseQuery({
    queryKey: ['ollama', 'status'],
    queryFn: getApiV1OllamaStatus,
    refetchInterval: 5000,
  })

  const startOllama = useMutation({
    mutationFn: postApiV1OllamaStart,
    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['ollama'] }),
  })

  const stopOllama = useMutation({
    mutationFn: postApiV1OllamaStop,
    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['ollama'] }),
  })

  return (
    <div className="space-y-4">
      <StatusRow
        label="Ollama"
        running={ollamaStatus?.running ?? false}
        onStart={startOllama.mutate}
        onStop={stopOllama.mutate}
      />
    </div>
  )
}
